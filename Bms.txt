class Validation
{
public:
    static bool validName(string name) {
        if (name.length() < 3 || name.length() > 20 || name[0] == ' ' || name[name.length() - 1] == ' ')
            return false;
        bool previousSpace = false;
        for (int i = 0; i < (int)name.length(); i++) {
            char c = name[i];
            if (c == ' ') {
                if (previousSpace) return false;
                previousSpace = true;
            }
            else if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z'))
                previousSpace = false;
            else
                return false;
        }
        return true;
    }
    static bool validPass(string pass) {
        if (pass.length() < 8 || pass.length() > 20)
            return false;
        for (int i = 0; i < pass.length(); i++) {
            if (pass[i] == ' ')
                return false;
        }
        return true;
    }
    static bool validBalance(double balance) {
        return balance >= 1500;
    }
    static bool validSalary(double salary) {
        return salary >= 5000;
    }
};

class Person
{
protected:
    int id;
    string name;
    string pass;
public:
    Person(int id = 0, string name = "", string pass = "") {
        this->id = id;
        this->name = name;
        this->pass = pass;
    }

    int getId() { return id; }
    string getName() { return name; }
    string getPass() { return pass; }

    void setId(int id) { this->id = id; }
    void setName(string name) {
        if (Validation::validName(name))
            this->name = name;
        else
            cout << "Invalid name." << endl;
    }
    void setPass(string pass) {
        if (Validation::validPass(pass))
            this->pass = pass;
        else
            cout << "Invalid password." << endl;
    }
    void display() {
        cout << "ID : " << id << endl;
        cout << "Name : " << name << endl;
        cout << "Password : " << pass << endl;
    }
};

class Client : public Person
{
private:
    double balance;
public:
    Client(int id = 0, string name = "", string pass = "", double balance = 1500)
        : Person(id, name, pass) {
        if (Validation::validBalance(balance))
            this->balance = balance;
        else
            this->balance = 1500;
    }

    double getBalance() { return balance; }
    void setBalance(double balance) {
        if (Validation::validBalance(balance))
            this->balance = balance;
        else
            cout << "Invalid balance." << endl;
    }
    void deposit(double amount) {
        if (amount <= 0) {
            cout << "Invalid amount." << endl;
            return;
        }
        balance += amount;
        cout << "Deposit successful." << endl;
    }
    void withdraw(double amount) {
        if (amount <= 0 || balance - amount < 1500) {
            cout << "Withdrawal failed." << endl;
            return;
        }
        balance -= amount;
        cout << "Withdrawal successful." << endl;
    }
    void transferTo(double amount, Client& recipient) {
        if (amount <= 0 || balance - amount < 1500) {
            cout << "Transfer failed." << endl;
            return;
        }
        balance -= amount;
        recipient.balance += amount;
        cout << "Transfer successful." << endl;
    }
    void display() {
        Person::display();
        cout << "Balance : " << balance << endl;
    }
};

class Parser
{
public:
    static vector<string> split(string line) {
        vector<string> data;
        string word;
        // Read both the old format (&) and the new format (-).
        char separator = (line.find('&') != string::npos) ? '&' : '-';
        for (int i = 0; i < (int)line.length(); i++) {
            if (line[i] == separator) {
                data.push_back(word);
                word = "";
            }
            else {
                word += line[i];
            }
        }
        data.push_back(word);
        return data;
    }
    static Client parseToClient(string line) {
        vector<string> d = split(line);
        if (d.size() != 4) return Client();
        return Client(stoi(d[0]), d[1], d[2], stod(d[3]));
    }
    static Employee parseToEmployee(string line) {
        vector<string> d = split(line);
        if (d.size() != 4) return Employee();
        return Employee(stoi(d[0]), d[1], d[2], stod(d[3]));
    }
    static Admin parseToAdmin(string line) {
        vector<string> d = split(line);
        if (d.size() != 4) return Admin();
        return Admin(stoi(d[0]), d[1], d[2], stod(d[3]));
    }
};

class FilesHelper
{
public:
    static void saveLast(string fileName, int id) {
        ofstream file(fileName);
        file << id;
        file.close();
    }
    static int getLast(string fileName) {
        int id = 0;
        ifstream file(fileName);
        file >> id;
        file.close();
        return id;
    }
    static void saveClient(Client c) {
        ofstream file("Clients.txt", ios::app);
        file << c.getId() << "-"
            << c.getName() << "-"
            << c.getPass() << "-"
            << c.getBalance() << endl;
        file.close();
        saveLast("ClientLastId.txt", c.getId());
    }
    static void saveEmployee(string fileName, string lastIdFile, Employee e) {
        ofstream file(fileName, ios::app);
        file << e.getId() << "-"
            << e.getName() << "-"
            << e.getPass() << "-"
            << e.getSalary() << endl;
        file.close();
        saveLast(lastIdFile, e.getId());
    }
    static void saveAdmin(Admin a) {
        ofstream file("Admins.txt", ios::app);
        file << a.getId() << "-"
            << a.getName() << "-"
            << a.getPass() << "-"
            << a.getSalary() << endl;
        file.close();
        saveLast("AdminLastId.txt", a.getId());
    }
    static void getClients() {
        allClients.clear();
        ifstream file("Clients.txt");
        string line;
        while (getline(file, line)) {
            if (line != "") {
                try { allClients.push_back(Parser::parseToClient(line)); }
                catch (...) { cout << "Skipping invalid client record in Clients.txt" << endl; }
            }
        }
        file.close();
    }
    static void getEmployees() {
        allEmployees.clear();
        ifstream file("Employees.txt");
        string line;
        while (getline(file, line)) {
            if (line != "") {
                try { allEmployees.push_back(Parser::parseToEmployee(line)); }
                catch (...) { cout << "Skipping invalid employee record in Employees.txt" << endl; }
            }
        }
        file.close();
    }
    static void getAdmins() {
        allAdmins.clear();
        ifstream file("Admins.txt");
        string line;
        while (getline(file, line)) {
            if (line != "") {
                try { allAdmins.push_back(Parser::parseToAdmin(line)); }
                catch (...) { cout << "Skipping invalid admin record in Admins.txt" << endl; }
            }
        }
        file.close();
    }
    static void clearFile(string fileName, string lastIdFile) {
        ofstream file(fileName);
        file.close();
        saveLast(lastIdFile, 0);
    }
    static void saveAllClients() {
        ofstream file("Clients.txt");
        for (int i = 0; i < allClients.size(); i++) {
            file << allClients[i].getId() << "-"
                << allClients[i].getName() << "-"
                << allClients[i].getPass() << "-"
                << allClients[i].getBalance() << endl;
        }
        file.close();
    }
    static void saveAllEmployees() {
        ofstream file("Employees.txt");
        for (int i = 0; i < allEmployees.size(); i++) {
            file << allEmployees[i].getId() << "-"
                << allEmployees[i].getName() << "-"
                << allEmployees[i].getPass() << "-"
                << allEmployees[i].getSalary() << endl;
        }
        file.close();
    }
    static void saveAllAdmins() {
        ofstream file("Admins.txt");
        for (int i = 0; i < allAdmins.size(); i++) {
            file << allAdmins[i].getId() << "-"
                << allAdmins[i].getName() << "-"
                << allAdmins[i].getPass() << "-"
                << allAdmins[i].getSalary() << endl;
        }
        file.close();
    }
};
